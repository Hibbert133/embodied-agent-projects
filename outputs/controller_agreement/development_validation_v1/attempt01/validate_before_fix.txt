"""Source-bound pure checks and exact owned-index validation before TRAIN read."""
from pathlib import Path
from time import perf_counter_ns
import subprocess,sys,unittest
ROOT=Path(__file__).resolve().parents[1];sys.path.insert(0,str(ROOT))
from scripts import run_controller_agreement_v1 as entry
from src.analysis.release_learning_guard_v1 import read_only
from src.analysis.response_action_io_v1 import verify_runtime
from outputs.information_release.validation_v1.validate_freeze import index_bytes
from outputs.research_archive.preflight_v1.scan_snapshot import EXTRA
io=entry.io


def main():
    start=perf_counter_ns();out=entry.BASE/'freeze_validation_v1';out.mkdir(parents=True,exist_ok=False)
    io.require(not io.git('diff','--name-only') and not io.git('diff','--cached','--name-only'),'old tracked sources and index unchanged')
    io.require(io.read(entry.CONFIG)==entry.config(),'exact zero-interaction contract')
    prior_path=ROOT/'outputs/graybox_qualification/result_validation_v2/source_reuse.json'
    freeze_path=ROOT/'outputs/graybox_qualification/freeze_validation_v2/complete.json'
    prior=io.read(prior_path);freeze=io.read(freeze_path)
    io.require(prior['integrity']==freeze['integrity']=='PASSED' and prior['unique_tests']==freeze['unique_test_total']==2428,'prior complete source-bound checks')
    old={**prior['source_sha256'],**freeze['source_sha256']}
    for n,h in old.items():io.require(io.sha(ROOT/n)==h,'unchanged previously tested source '+n)
    verify_runtime(io.read(ROOT/'outputs/correction_scope/freeze_validation_v1/runtime_lock.json'))
    before={n:io.sha(ROOT/n) for n in (*entry.OWN,*entry.DOCS)};guard={};ids=[]
    with (out/'tests.log').open('x',encoding='utf8',newline='\n') as log,read_only(guard):
        suite=unittest.defaultTestLoader.loadTestsFromName('tests.test_controller_agreement_v1')
        def collect(s):
            for t in s:
                if isinstance(t,unittest.TestSuite):collect(t)
                else:ids.append(t.id())
        collect(suite);result=unittest.TextTestRunner(stream=log,verbosity=2).run(suite)
    io.write(out/'tests.json',dict(tests=result.testsRun,ids=ids,success=result.wasSuccessful(),guard=guard,source_sha256=before))
    io.require(result.wasSuccessful() and result.testsRun==len(set(ids))==20,'20 pure no-physics tests')
    io.write(out/'source_reuse.json',dict(integrity='PASSED',unique_tests=2448,new_unique_tests=20,reused_unique_tests=2428,
        source_sha256=old,prior_receipt_sha256={p.relative_to(ROOT).as_posix():io.sha(p) for p in (prior_path,freeze_path)},old_simulation_tests_repeated=False))
    io.write(out/'source_contract.json',entry.source_contract())
    def run(name,args):
        t=perf_counter_ns();p=subprocess.run(args,cwd=ROOT,capture_output=True,text=True,encoding='utf8',errors='replace')
        io.write(out/name,dict(args=args,returncode=p.returncode,stdout=p.stdout,stderr=p.stderr,wall_ns=perf_counter_ns()-t));io.require(p.returncode==0,name)
    names=set(entry.OWN)|set(entry.DOCS)
    def stage(label):
        nonlocal names
        names|={p.relative_to(ROOT).as_posix() for p in out.iterdir() if p.is_file()}
        path=out/(label+'.nul');names.add(path.relative_to(ROOT).as_posix());path.write_bytes(b'\0'.join(n.encode() for n in sorted(names))+b'\0')
        io.git('-c','maintenance.auto=false','-c','gc.auto=0','add','--pathspec-from-file='+path.relative_to(ROOT).as_posix(),'--pathspec-file-nul')
    def check():
        actual=index_bytes(names)
        for n in actual:
            data=(ROOT/n).read_bytes()
            if b'\0' not in data:
                for label,regex in EXTRA.items():io.require(regex.search(data.decode('utf8',errors='replace')) is None,'actual index credential '+label+' '+n)
        io.require(not io.git('diff','--check') and not io.git('diff','--cached','--check'),'ordinary and staged differences');return actual
    run('compile.json',[sys.executable,'-m','compileall','-q','src','scripts','tests']);run('dependencies.json',[sys.executable,'-m','pip','check'])
    stage('owned_initial');run('tracked_secrets.json',[sys.executable,'scripts/check_tracked_secrets.py'])
    stage('owned_checked');actual=check();io.write(out/'index_disk_proof.json',dict(integrity='PASSED',object_sha256=actual,basic_and_extra_actual_index_scan=True))
    for n,h in before.items():io.require(io.sha(ROOT/n)==h,'tested source changed '+n)
    io.write(out/'complete.json',dict(integrity='PASSED',source_sha256=before,unique_test_total=2448,new_tests=20,reused_tests=2428,guard=guard,
        science_started=False,new_physics=0,new_fits=0,online_LLM_API_calls=0,new_policy_queries=0,
        receipt_sha256={p.name:io.sha(p) for p in out.iterdir() if p.is_file()},wall_ns=perf_counter_ns()-start,publication_goal_complete=False))
    stage('owned_final');check();print(dict(integrity='PASSED',objects=len(names),unique_tests=2448,science='NOT_STARTED'),flush=True)


if __name__=='__main__':main()
